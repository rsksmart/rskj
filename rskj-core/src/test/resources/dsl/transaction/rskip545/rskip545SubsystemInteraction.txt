// RSKIP-545 interaction matrix cells that need a real chain:
//   - Bridge / PowPeg: pegout from a delegated account vs from a contract vs from a plain EOA
//   - Bridge / PowPeg: a Bridge refund crediting a delegated account must not run the delegate code
//   - Bridge / PowPeg: delegation *to* the Bridge address resolves to empty code
//   - REMASC: fee accounting for a block carrying a Type 4 transaction
//   - Nonce semantics: self-sponsored tuple carrying the pre-increment sender nonce is skipped
// The Java driver activates RSKIP-543/545/546 at block 0.
// 0x...01000006 is PrecompiledContracts.BRIDGE_ADDR.

# ============================================================================
# b01: peg-out requested by a Type 4 transaction whose sender delegates itself
# ============================================================================

transaction_build txPegoutFromDelegatedSender
    sender acc1
    nonce 0
    receiverAddress 0000000000000000000000000000000001000006
    value 1000000000000000000
    gas 500000
    maxPriorityFeePerGas 1000000000
    maxFeePerGas 2000000000
    transactionType 04
    authorization acc1 markerImpl 1
    build

block_build b01
    parent g00
    gasLimit 6800000
    transactions txPegoutFromDelegatedSender
    build

block_connect b01
assert_best b01
assert_tx_success txPegoutFromDelegatedSender
