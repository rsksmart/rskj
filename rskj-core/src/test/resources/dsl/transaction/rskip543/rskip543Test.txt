// RSKIP543 Consolidated DSL Test: EIP-2718 Style Typed Transactions in Rootstock
//
// Tests RSKIP543 envelope support without RSK-namespace Type 2 (fail-closed until a
// subtype RSKIP defines the payload):
//   1. Legacy transactions (Type 0x00) - no type prefix, first byte >= 0xc0
//   2. Standard EIP-2718 typed transactions (Type 0x01) - single byte prefix
//   3. Mixed legacy + Type 1 in the same block
//   4. Backward compatibility with legacy transactions

end

# Create test accounts with sufficient balance
account_new acc1 100000000000000000
account_new acc2 1000000
account_new acc3 1000000

# ============================================================================
# Test 1: Legacy Transaction (Type 0x00) - Backward Compatibility
# ============================================================================

transaction_build txLegacy
    sender acc1
    receiver acc2
    value 1000
    gas 21000
    build

block_build b01
    parent g00
    transactions txLegacy
    build

block_connect b01

assert_best b01
assert_tx_success txLegacy

# ============================================================================
# Test 2: Type 1 Transaction
# ============================================================================

transaction_build txType1
    sender acc1
    nonce 1
    receiver acc2
    value 2000
    gas 21000
    transactionType 01
    build

block_build b02
    parent b01
    transactions txType1
    build

block_connect b02

assert_best b02
assert_tx_success txType1

# ============================================================================
# Test 3: Mixed Transaction Types in Same Block (legacy + Type 1)
# ============================================================================

transaction_build txMixed1
    sender acc1
    nonce 2
    receiver acc2
    value 500
    gas 21000
    build

transaction_build txMixed2
    sender acc1
    nonce 3
    receiver acc2
    value 600
    gas 21000
    transactionType 01
    build

block_build b03
    parent b02
    transactions txMixed1 txMixed2
    build

block_connect b03

assert_best b03
assert_tx_success txMixed1
assert_tx_success txMixed2

# ============================================================================
# Final Assertions
# ============================================================================

assert_best b03
